#11주차 서포트 벡터머신 실습

library('e1071')
library('caret')
library('pROC')

df <- read.csv("C:/Users/USER/Downloads/TalkFile_telecom_churn.csv.csv")

df$Churn <- as.factor(df$Churn)

trainIndex <- createDataPartition(df$Churn,p = 0.8, list = FALSE)
train_data <- df[trainIndex,]
test_data <- df[-trainIndex,]

cost_value <- 1
gamma_value <- 0.1

svm_model <- svm(train_data$Churn ~., data = train_data,
                 type = 'C-classification',
                 kernel = 'polynomial',
                 cost = cost_value,
                 gamma = gamma_value,
                 probability = TRUE)

summary(svm_model)

predicted_probs <- predict(svm_model, test_data,probability =  TRUE)
predicted_probs

predicted_probs <- attr(predicted_probs, "probabilities")[,"Yes"]

predicted_classes <- ifelse(predicted_probs > 0.5,"Yes","No")
predicted_classes <- as.factor(predicted_classes)
predicted_classes

conf_matrix <- confusionMatrix(predicted_classes, test_data$Churn)
print(conf_matrix)

roc_curve <- roc(test_data$Churn, predicted_probs)
plot(roc_curve)
auc(roc_curve)
