뇌풀기용 실습 예제(시험 X)
library(ggplot2)

df<- data.frame(
  group = c(rep("Group1",60),rep("Group2",60)),
  values = c(rnorm(60,mean = 0,sd = 1),rnorm(60,mean=2,sd=1)))

ggplot(df,aes(x=group,y=values))+
  geom_boxplot(fill = c("lightblue","lightgreen"),outlier.color = "red")+
  labs(title = "Boxplot Example")+
  xlab("Group")+
  ylab("Values")
